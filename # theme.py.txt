# theme.py
import streamlit as st
import plotly.io as pio
import plotly.graph_objects as go

# JPC Brand Palette
brand_palette = {
    "brand_primary": "#0A1A2A",    # Deep Navy/Charcoal
    "brand_accent": "#C5A059",     # Warm Metallic Gold
    "bg": "#F9F9F7",               # Off-white/Bone neutral
    "surface": "#FFFFFF",          # Card background
    "text_primary": "#1A1A1A",     # Accessible dark grey
    "border": "#E0E0DB",           # Subtle border
    "status_good": "#287D3C",      # Forest Green
    "status_warning": "#B7791F",   # Ochre
    "status_bad": "#C53030",       # Crimson
}

def apply_theme():
    # Inject Global CSS
    st.markdown(f"""
    <style>
        :root {{
            --primary: {brand_palette['brand_accent']};
            --bg: {brand_palette['bg']};
        }}
        .stApp {{ background-color: {brand_palette['bg']}; color: {brand_palette['text_primary']}; }}
        h1, h2, h3 {{ color: {brand_palette['brand_primary']} !important; font-family: 'Georgia', serif; }}
        [data-testid="stMetric"] {{
            background-color: {brand_palette['surface']}; 
            border: 1px solid {brand_palette['border']};
            border-radius: 8px; padding: 1.5rem;
        }}
        .stButton>button {{
            background-color: {brand_palette['brand_accent']}; color: white; border: none;
            border-radius: 4px; font-weight: 600; padding: 0.6rem 2rem;
        }}
        [data-testid="stSidebar"] {{ background-color: {brand_palette['brand_primary']}; }}
        [data-testid="stSidebar"] * {{ color: white !important; }}
    </style>
    """, unsafe_allow_html=True)

    # Global Plotly Template
    template = go.layout.Template()
    template.layout = {
        "paper_bgcolor": "rgba(0,0,0,0)",
        "plot_bgcolor": "rgba(0,0,0,0)",
        "font": {"color": brand_palette['text_primary'], "family": "Arial"},
        "colorway": [brand_palette['brand_accent'], "#5C6B7A", "#A0AEC0"],
    }
    pio.templates["jpc_luxury"] = template
    pio.templates.default = "jpc_luxury"
