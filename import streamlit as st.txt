# --- MODULE 2: SCORING ENGINE (Math Model) ---
def score_engine(val, target, norm_type, observed_at, confidence=0.7):
    # Normalization
    if norm_type == "HIGHER_BETTER":
        s = min(val / target, 1.0) * 100
    else: # LOWER_BETTER
        s = max(0, 1 - (val / target)) * 100
    
    # Time Decay (Stale data penalty)
    age = (datetime.now() - observed_at).days
    decay = (age // 30) * 5
    
    # Final Score with Confidence weighting
    return max(0, (s * (0.5 + 0.5 * confidence)) - decay)
