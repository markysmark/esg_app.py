# 3. MODULE 1 & 2: OBLIGATIONS & SCORING ENGINES
def calculate_indicator_score(val, target, norm_type, date_observed, confidence=0.8):
    # Normalization logic
    if norm_type == "HIGHER_BETTER":
        base_s = min(val / target, 1.0) * 100
    else:
        base_s = max(0, 1 - (val / target)) * 100
    
    # Time Decay (Penalty: -5 pts per 30 days)
    days_old = (datetime.now() - date_observed).days
    decay = (days_old // 30) * 5
    
    # Evidence confidence penalty (Alpha=0.5)
    return max(0, (base_s * (0.5 + 0.5 * confidence)) - decay)
