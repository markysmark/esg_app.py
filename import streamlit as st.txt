# --- 2. ENGINE CORE: SCORING & PDF ---
def calculate_indicator_score(val, target, date_observed, confidence=0.8):
    if val <= 0: return 35  # Penalty for Missing Data
    base_s = min(val / target, 1.0) * 100
    age = (datetime.now() - date_observed).days
    decay = (age // 30) * 5 # -5 pts per month
    return max(0, (base_s * (0.5 + 0.5 * confidence)) - decay)
