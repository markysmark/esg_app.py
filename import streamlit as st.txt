def calculate_esg_score(val, target, date_observed, confidence=0.8):
    """0-100 Score with Time-Decay (-5pts/30 days) and Missing Penalties."""
    if val <= 0: return 35 # Missing Data Penalty
    base = min(val / target, 1.0) * 100
    age = (datetime.now() - date_observed).days
    decay = (age // 30) * 5
    return max(0, (base * (0.5 + 0.5 * confidence)) - decay)
